```python
from pathlib import Path
import csv, hashlib, io

source = Path("docs/独立验收/02-玩法验收/任务/V002-夹具复验.md").read_text()
data = source.split("```csv\n", 1)[1].split("```", 1)[0]
rows = list(csv.DictReader(io.StringIO(data)))
print(hashlib.sha256(data.encode()).hexdigest())
print(len({(r["回法"], r["左手"], r["偏移ms"]) for r in rows}))
print(max(float(r["拍头误差cm"]) for r in rows))
for row in rows:
    if row["过网"] == "false" or (
        row["回法"] == "反手平抽" and row["偏移ms"] in ("-100", "-50")
    ):
        print(row["回法"], row["左手"], row["偏移ms"], row["拍头误差cm"],
              row["握点跳变cm"], row["过网"])
```
