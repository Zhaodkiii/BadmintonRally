```sh
t=$(mktemp -d)

cat > "$t/NoButton.swift" <<'SWIFT'
import Foundation
@main enum NoButton {
  static func main() {
    for left in [false,true] {
      for fps in [30,60,120] {
        var game = Rally()
        game.leftHanded = left
        game.phase = .flight
        game.flight = Flight(from:Position(x:0.10,y:0.51,z:0.45),to:Position(x:0.14,y:0.77),start:0,duration:0.8,arc:0.28,incoming:true,shot:.drive)
        let inbound = game.flight!
        for _ in 0..<(3*fps) {
          game.previewSwing(SwingImpact(power:0.9,sweep:0.8,verticalSweep:0.15))
          game.tick(1.0/Double(fps))
          precondition(game.returns == 0 && game.hits == 0 && game.judgedContactTime == nil)
          if game.phase == .point { break }
          let expected = inbound.at(game.now), actual = game.flight!.at(game.now)
          precondition(game.flight!.incoming && actual.x == expected.x && actual.y == expected.y && actual.z == expected.z)
        }
        precondition(game.pointCount == 1 && game.score == [0,1] && !game.playerServes && game.message == "来球落地")
        print(left, fps, game.returns, game.score, game.message)
      }
    }
  }
}
SWIFT
swiftc Clients/iOS/Game/Rally.swift "$t/NoButton.swift" -o "$t/no-button" && "$t/no-button"
```
